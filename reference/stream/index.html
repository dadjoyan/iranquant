    AsyncHTTPClient ¶

    AsyncHTTPClient(
    *,
    timeout: float = DEFAULT_TIMEOUT,
    max_concurrency: int = MAX_CONCURRENCY,
    max_retries: int = MAX_RETRIES,
    retry_wait: float = RETRY_WAIT,
    headers: Mapping[str, str] | None = None,
    proxy: str | None = None,
    trust_env: bool | None = None,
)

    Small retrying wrapper around :class:httpx.AsyncClient.

    Three transports are maintained:

    • a regular verified client used for TSETMC;
    • an insecure client used for every *.codal.ir domain, whose certificate chains frequently fail strict verification;
    • an insecure HTTP/1.1 fallback used once for TSETMC when the primary connection fails with an SSL/proxy-style error.

    Environment proxy variables are ignored by default (trust_env=False) to avoid corporate/VPN HTTP proxies breaking direct connections to domestic Iranian hosts. Pass proxy=... to opt into an explicit proxy.

    Create the pooled HTTP transports.

    def __init__(
    self,
    *,
    timeout: float = DEFAULT_TIMEOUT,
    max_concurrency: int = MAX_CONCURRENCY,
    max_retries: int = MAX_RETRIES,
    retry_wait: float = RETRY_WAIT,
    headers: Mapping[str, str] | None = None,
    proxy: str | None = None,
    trust_env: bool | None = None,
) -> None:
    """Create the pooled HTTP transports.

    Args:
        timeout: Per-request timeout in seconds.  Defaults to
            :data:`iranquant.config.DEFAULT_TIMEOUT` (30 s).
        max_concurrency: Maximum number of simultaneous requests enforced
            by an internal :class:`asyncio.Semaphore`.  Defaults to
            :data:`iranquant.config.MAX_CONCURRENCY` (10).
        max_retries: Total attempts per request.  Defaults to
            :data:`iranquant.config.MAX_RETRIES` (3).
        retry_wait: Base wait used by the exponential retry backoff, in
            seconds.  Defaults to :data:`iranquant.config.RETRY_WAIT` (0.5).
        headers: Extra headers merged over the built-in TSETMC/Codal
            headers.
        proxy: Optional explicit proxy URL, e.g.
            ``"http://127.0.0.1:9090"``.  When omitted, environment proxy
            variables are ignored.
        trust_env: Whether to honour ``HTTP_PROXY`` / ``HTTPS_PROXY`` and
            other environment settings.  Defaults to ``True`` only when an
            explicit ``proxy`` is supplied, otherwise ``False``.

    Examples:
        >>> client = AsyncHTTPClient(timeout=10.0, max_concurrency=5)
        >>> client.trust_env
        False
        >>> client = AsyncHTTPClient(proxy="http://127.0.0.1:9090")
        >>> client.trust_env
        True
    """
    self._semaphore = asyncio.Semaphore(max_concurrency)
    self._max_retries = max_retries
    self._retry_wait = retry_wait
    self._headers = dict(headers or {})
    self.proxy = proxy
    # Explicit proxies imply honouring the environment unless overridden.
    self.trust_env = (proxy is not None) if trust_env is None else trust_env

    common: dict[str, Any] = {
        "timeout": timeout,
        "follow_redirects": True,
        "trust_env": self.trust_env,
        "http2": False,
    }
    if proxy is not None:
        common["proxy"] = proxy

    self._client = httpx.AsyncClient(
        headers={**TSETMC_HEADERS, **self._headers},
        **common,
    )
    # Dedicated insecure transport for Codal (SSL chain bugs).
    self._codal_client = httpx.AsyncClient(
        headers={**CODAL_HEADERS, **self._headers},
        verify=False,
        **common,
    )
    # Last-resort HTTP/1.1 + no-verify transport for TSETMC.
    self._fallback_client = httpx.AsyncClient(
        headers={**TSETMC_HEADERS, **self._headers},
        verify=False,
        **common,
    )

    async def get(
    self,
    url: str,
    *,
    params: Mapping[str, Any] | None = None,
    headers: Mapping[str, str] | None = None,
    raw: bool = False,
) -> httpx.Response:
    """Perform a GET request with retries and TLS fallbacks.

    The request is retried up to ``max_retries`` times on network errors,
    and responses with status codes >= 500 are treated as retryable.  A
    TSETMC connection that fails with ``SSLError``/``ConnectError`` is
    retried once over an insecure HTTP/1.1 transport.  Every ``*.codal.ir``
    request uses the dedicated certificate-relaxed transport.

    Args:
        url: Absolute request URL.
        params: Optional query parameters.  Ignored for URL encoding when
            ``raw`` is true.
        headers: Optional per-request headers merged over the defaults.
        raw: When true the URL string is sent verbatim without
            percent-encoding.  Required for the MarketWatch endpoint, whose
            ``paperTypes[0]=1`` brackets must stay literal.

    Returns:
        The successful :class:`httpx.Response`.

    Raises:
        NetworkError: If the host is unreachable, a non-2xx status is
            returned, or all retries are exhausted.

    Examples:
        >>> response = await client.get("https://cdn.tsetmc.com/api/BestLimits/1")
        >>> response.status_code
        200
    """

    target = self._build_url(url, params, raw=raw)

    try:
        async for attempt in self._retrying():
            with attempt:
                response = await self._send_with_fallback(target, headers)
                if response.status_code >= 500:
                    raise NetworkError(f"Server error {response.status_code} for {target}")
                response.raise_for_status()
                return response
    except httpx.HTTPStatusError as exc:
        raise NetworkError(f"HTTP {exc.response.status_code} for {exc.request.url}") from exc
    except httpx.HTTPError as exc:
        raise NetworkError(f"Request failed for {target}: {exc}") from exc

    raise NetworkError(f"Request failed after retries: {target}")

    async def get_json(
    self,
    url: str,
    *,
    params: Mapping[str, Any] | None = None,
    headers: Mapping[str, str] | None = None,
    raw: bool = False,
) -> Any:
    """GET a URL and decode its JSON body.

    Args:
        url: Absolute request URL.
        params: Optional query parameters.
        headers: Optional per-request headers.
        raw: When true, send the URL untouched (see :meth:`get`).

    Returns:
        The decoded JSON value (usually a ``dict`` or ``list``).

    Raises:
        NetworkError: If the request fails after retries.
        ParseError: If the response body is not valid JSON.

    Examples:
        >>> payload = await client.get_json(
        ...     "https://cdn.tsetmc.com/api/Instrument/GetInstrumentSearch/فولاد"
        ... )
        >>> "instrumentSearch" in payload
        True
    """

    response = await self.get(url, params=params, headers=headers, raw=raw)
    try:
        return response.json()
    except ValueError as exc:
        raise ParseError(f"Invalid JSON from {response.request.url}") from exc

    MarketWatcher ¶

    MarketWatcher(client: AsyncHTTPClient | None = None)

    Poll MarketWatch and emit queue/volume transitions.

    process_snapshot is synchronous and stateful so it can also be used in backtests or tests without any network access. The state machine emits:

    • BUY_QUEUE_LOCKED / BUY_QUEUE_BROKEN when close >= tmax with a positive buy queue starts or stops.
    • SELL_QUEUE_LOCKED / SELL_QUEUE_BROKEN when close <= tmin with a positive sell queue starts or stops.
    • VOLUME_SPIKE when the per-poll volume increase is at least 1.8x the running average increase.

    def on_event(
    self,
    event_type: str,
) -> Callable[[EventCallback], EventCallback]:
    """Register a sync or async callback for an event type (``"*"`` = all).

    Args:
        event_type: A specific event constant or ``"*"`` to receive every
            event.

    Returns:
        A decorator that registers the callback and returns it unchanged,
        so the method can be used as ``@watcher.on_event(...)``.

        ...     print(event.event_type, event.symbol)

        self._listeners.setdefault(event_type, []).append(callback)
        return callback

    return decorator

    async def dispatch(self, events: Iterable[MarketEvent]) -> None:
    """Invoke every matching listener for each event.

    Sync callbacks are called directly; awaitable results are awaited.
    Listeners registered for ``"*"`` run after type-specific listeners.

    Args:
        events: Events to deliver, typically returned by
            :meth:`process_snapshot`.

    Examples:
        >>> await watcher.dispatch(watcher.process_snapshot(snapshot))
    """
    for event in events:
        callbacks = [
            *self._listeners.get(event.event_type, []),
            *self._listeners.get("*", []),
        ]
        for callback in callbacks:
            result = callback(event)
            if inspect.isawaitable(result):
                await cast(Awaitable[Any], result)

    def process_snapshot(
    self,
    snapshot: DataFrame,
    *,
    timestamp: dt.datetime | None = None,
    symbols: list[str] | None = None,
) -> list[MarketEvent]:
    """Update internal state from a MarketWatch frame and return events.

    No I/O is performed; the result is deterministic for a given snapshot
    sequence, so this is safe to use in backtests and unit tests.

    Args:
        snapshot: A MarketWatch frame (from
            :func:`iranquant.market_watch`) with at least ``close``,
            ``tmax``, ``tmin`` and ``volume``.  ``ins_code``, ``symbol``
            and ``order_book`` are used when present.
        timestamp: Timestamp attached to emitted events.  Defaults to the
            current UTC time.
        symbols: Optional allow-list.  Numeric strings are matched against
            ``ins_code``; other strings are normalized and matched against
            ``symbol``.

    Returns:
        The newly triggered events in snapshot order.  An empty frame
        returns an empty list without mutating state.

    Examples:
        >>> events = watcher.process_snapshot(iq.market_watch())
        >>> [event.event_type for event in events]
        ['BUY_QUEUE_LOCKED']
    """

    frame = to_pandas(snapshot)
    if frame.empty:
        return []

    wanted_codes = {str(symbol) for symbol in symbols or [] if str(symbol).isdigit()}
    wanted_symbols = {
        normalize_persian(symbol) for symbol in symbols or [] if not str(symbol).isdigit()
    }
    moment = timestamp or dt.datetime.now(dt.timezone.utc)

    events: list[MarketEvent] = []
    for _, row in frame.iterrows():
        ins_code = str(row.get("ins_code", "") or "")
        symbol = str(row.get("symbol", "") or "")
        if symbols is not None:
            matches = ins_code in wanted_codes or normalize_persian(symbol) in wanted_symbols
            if not matches:
                continue

        key = ins_code or symbol
        if not key:
            continue

        close = _to_float(row.get("close"))
        tmax = _to_float(row.get("tmax"))
        tmin = _to_float(row.get("tmin"))
        volume = _to_float(row.get("volume"))
        buy_volume = _book_volume(row.get("order_book"), "buy")
        sell_volume = _book_volume(row.get("order_book"), "sell")

        buy_locked = tmax > 0 and close >= tmax and buy_volume > 0
        sell_locked = tmin > 0 and close <= tmin and sell_volume > 0

        state = self._states.get(key)
        is_new = state is None
        if state is None:
            state = _SymbolState()
            self._states[key] = state

        base_data: dict[str, Any] = {
            "close": close,
            "tmax": tmax,
            "tmin": tmin,
            "buy_queue_volume": buy_volume,
            "sell_queue_volume": sell_volume,
            "volume": volume,
        }

        if buy_locked and not state.buy_locked:
            events.append(
                MarketEvent(BUY_QUEUE_LOCKED, ins_code, symbol, moment, dict(base_data))
            )
        elif state.buy_locked and not buy_locked:
            events.append(
                MarketEvent(BUY_QUEUE_BROKEN, ins_code, symbol, moment, dict(base_data))
            )

        if sell_locked and not state.sell_locked:
            events.append(
                MarketEvent(SELL_QUEUE_LOCKED, ins_code, symbol, moment, dict(base_data))
            )
        elif state.sell_locked and not sell_locked:
            events.append(
                MarketEvent(SELL_QUEUE_BROKEN, ins_code, symbol, moment, dict(base_data))
            )

        if not is_new and state.last_volume is not None:
            delta = volume - state.last_volume
            if delta < 0:
                # A new trading day/reset: restart the spike baseline.
                state.avg_delta = 0.0
                state.delta_count = 0
            elif delta > 0:
                previous_average = state.avg_delta
                if state.delta_count > 0 and delta >= 1.8 * previous_average:
                    events.append(
                        MarketEvent(
                            VOLUME_SPIKE,
                            ins_code,
                            symbol,
                            moment,
                            {
                                "volume": volume,
                                "delta": delta,
                                "avg_delta": previous_average,
                                "close": close,
                            },
                        )
                    )
                state.delta_count += 1
                state.avg_delta = (
                    (previous_average * (state.delta_count - 1)) + delta
                ) / state.delta_count

        state.buy_locked = buy_locked
        state.sell_locked = sell_locked
        state.last_volume = volume
        state.last_close = close

    return events

    async def watch(
    self,
    interval: float = 5.0,
    symbols: list[str] | None = None,
    only_market_hours: bool = False,
) -> AsyncGenerator[MarketEvent, None]:
    """Poll MarketWatch forever, yielding state-change events.

    Each iteration fetches a full market snapshot, updates the state
    machine, dispatches callbacks, yields every new event, then sleeps.
    IranQuant errors during a poll are swallowed and retried on the next
    interval.

    Args:
        interval: Seconds between full MarketWatch polls.  Must be positive.
            Defaults to ``5.0``.
        symbols: Optional allow-list passed to :meth:`process_snapshot`.
        only_market_hours: When ``True``, skip polling outside
            Saturday-Wednesday 09:00-12:30 Tehran time.  Defaults to
            ``False``.

    Yields:
        :class:`MarketEvent` objects as state transitions occur.

    Raises:
        ValueError: If ``interval <= 0``.

    Examples:
        >>> async for event in watcher.watch(interval=3.0, symbols=["فولاد"]):
        ...     print(event.event_type, event.symbol)
    """

    if interval <= 0:
        raise ValueError("interval must be positive")

    owns_client = self._client is None
    if self._client is None:
        self._client = AsyncHTTPClient()
    client = self._client

    try:
        while True:
            if only_market_hours and not is_market_open():
                await asyncio.sleep(interval)
                continue
            try:
                snapshot = await get_market_watch(client)
            except IranQuantError:
                await asyncio.sleep(interval)
                continue

            events = self.process_snapshot(snapshot, symbols=symbols)
            await self.dispatch(events)
            for event in events:
                yield event
            await asyncio.sleep(interval)
    finally:
        if owns_client:
            await client.aclose()
            self._client = None

    def _book_volume(order_book: Any, side: str) -> float:

    def is_market_open(now: dt.datetime | None = None) -> bool:
    """Return True during Saturday-Wednesday TSETMC sessions (09:00-12:30 IRST).

    Python weekday numbering is Monday=0, so the active days are 5, 6, 0, 1
    and 2.  The time is interpreted in ``Asia/Tehran`` (with a fixed +03:30

            Tehran local time.  Defaults to the current instant.

        local time is between 09:00 and 12:30 inclusive, otherwise ``False``.

    Examples:
        >>> import datetime as dt
        >>> is_market_open(dt.datetime(2026, 10, 31, 10, 0))  # Saturday

        False

    current = now or dt.datetime.now(timezone)
    if current.tzinfo is None:
        current = current.replace(tzinfo=timezone)
    current = current.astimezone(timezone)
    if current.weekday() not in {5, 6, 0, 1, 2}:
        return False
    current_time = current.time()
    return dt.time(9, 0) <= current_time <= dt.time(12, 30)
